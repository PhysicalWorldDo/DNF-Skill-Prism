cmake_minimum_required(VERSION 3.20)
project(DNFSkillPrismCore VERSION 1.1.4 LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The current resource reader uses Windows APIs. Build on Windows x64.")
endif()
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ZLIB REQUIRED)
find_path(PRISM_JSON_INCLUDE nlohmann/json.hpp REQUIRED)
add_library(dnflib STATIC
  native/dnflib/src/dnflib.c native/dnflib/src/dxt.c native/dnflib/src/img.c
  native/dnflib/src/io_helper.c native/dnflib/src/miniz.c native/dnflib/src/npk.c
  native/dnflib/src/palette.c native/dnflib/src/pixel.c)
target_include_directories(dnflib PUBLIC native/dnflib/include PRIVATE native/dnflib/src)
add_library(prism_core STATIC
  native/prism_core.cpp native/color.cpp native/patch.cpp native/catalog_match.cpp
  native/rep/protocol.cpp native/rep/resources.cpp native/rep/catalog.cpp)
target_include_directories(prism_core PUBLIC native "${PRISM_JSON_INCLUDE}")
target_compile_definitions(prism_core PUBLIC NOMINMAX UNICODE _UNICODE)
target_link_libraries(prism_core PUBLIC dnflib ZLIB::ZLIB)
if(MSVC)
  target_compile_options(prism_core PUBLIC /utf-8)
endif()
include(CTest)
if(BUILD_TESTING)
  foreach(name IN ITEMS core_tests preset_store_tests)
    add_executable(${name} tests/${name}.cpp)
    target_link_libraries(${name} PRIVATE prism_core)
    if(MINGW)
      target_link_options(${name} PRIVATE -static)
    endif()
    add_test(NAME ${name} COMMAND ${name})
  endforeach()
endif()
